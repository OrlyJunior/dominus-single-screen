cmake_minimum_required(VERSION 3.20)
project(DominusSingleScreen VERSION 0.2.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This development runtime requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(dominus_core STATIC src/core.cpp)
target_include_directories(dominus_core PUBLIC include)
add_library(dominus_graphics STATIC src/graphics.cpp)
target_link_libraries(dominus_graphics PUBLIC dominus_core)
add_library(dominus_marker STATIC src/marker_blink.cpp src/marker_identity.cpp)
target_link_libraries(dominus_marker PUBLIC dominus_graphics bcrypt)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include PRIVATE third_party/minhook/src)
add_library(d3d11 SHARED src/proxy.cpp src/native_runtime.cpp src/render_capture.cpp src/resource_observer.cpp src/input_observer.cpp src/live_controller.cpp src/title_identity.cpp src/pointer_remap.cpp src/d3d11.def)
target_link_libraries(d3d11 PRIVATE dominus_marker minhook bcrypt user32 gdi32 ole32)
set_target_properties(d3d11 PROPERTIES PREFIX "" OUTPUT_NAME d3d11)
add_executable(attach_observer tools/attach_observer.cpp)
if(MINGW)
  target_link_options(attach_observer PRIVATE -municode)
endif()
foreach(target dominus_core dominus_graphics dominus_marker minhook d3d11)
  target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0602)
  if(NOT MSVC)
    target_compile_options(${target} PRIVATE -fms-extensions)
  endif()
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE dominus_core)
add_test(NAME core COMMAND core_tests)
add_executable(steam_pad_tests tests/steam_pad_tests.cpp src/input_observer.cpp)
target_link_libraries(steam_pad_tests PRIVATE dominus_core minhook)
if(NOT MSVC)
  target_compile_options(steam_pad_tests PRIVATE -fms-extensions)
endif()
# Requires an explicitly supplied user-owned analysis executable; not a default CTest test.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/d3d11_tests.cpp")
  add_executable(dominus_d3d11_tests tests/d3d11_tests.cpp src/title_identity.cpp)
  target_link_libraries(dominus_d3d11_tests PRIVATE dominus_marker user32 gdi32 ole32)
  if(NOT MSVC)
    target_compile_options(dominus_d3d11_tests PRIVATE -fms-extensions)
  endif()
  add_test(NAME d3d11_synthetic COMMAND dominus_d3d11_tests --proxy "$<TARGET_FILE:d3d11>")
endif()
